cmake_minimum_required(VERSION 3.20)
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "CANN toolkit root")
if(NOT ASCEND_CANN_PACKAGE_PATH)
    message(FATAL_ERROR "Pass -DASCEND_CANN_PACKAGE_PATH=<CANN 9 toolkit root>")
endif()
set(ENV{ASCEND_HOME_PATH} "${ASCEND_CANN_PACKAGE_PATH}")
set(CMAKE_ASC_ARCHITECTURES dav-2201 CACHE STRING "Ascend910B architecture")
find_package(ASC REQUIRED PATHS "${ASCEND_CANN_PACKAGE_PATH}/lib64/cmake" NO_DEFAULT_PATH)
project(hypic_tree_single_native LANGUAGES ASC CXX)
set_source_files_properties(../tree_single_kernel.cpp PROPERTIES LANGUAGE ASC)
add_library(hypic_tree_single_kernel SHARED ../tree_single_kernel.cpp)
target_compile_options(hypic_tree_single_kernel PRIVATE -O2 --npu-arch=dav-2201)
set_target_properties(hypic_tree_single_kernel PROPERTIES ASC_STANDARD 17)
